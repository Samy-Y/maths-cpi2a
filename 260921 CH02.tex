\chapter{Codimension}

\begin{proposition}
    Soit $S$ et $S'$ deux supplémentaires d'un sous-espace vectoriel $F$ de $E$.
    
    $$E = F \oplus S = F \oplus S'$$
    Alors $S \simeq S'$ (isomorphisme).
\end{proposition}

\begin{proof}
    Construisons un isomorphisme entre $S$ et $S'$. Pour ça, après analyse rapide, on peut définir l'application suivante :
    \begin{align*}
        u : S &\longrightarrow S' \\
        x &\longmapsto p(x)
    \end{align*}
    où $p$ est la projection de $S'$ parallèlement à $F$.
    Comme $E = F \oplus S'$, les images de $E$ par $p$ peuvent être écrire de la manière suivante :
    $$p : \begin{matrix} E = F \oplus S' & \longrightarrow & S' \\ f + s' & \longmapsto & s' \end{matrix}$$
    Et comme $E = F \oplus S$, les images de $E$ par $p$ peuvent être écrire de la manière suivante :
    $$p : \begin{matrix} E = F \oplus S & \longrightarrow & S' \\ f + s & \longmapsto & s' \end{matrix}$$
    Il est facile de vérifier que $u$ est bien linéaire. Il reste à montrer que $u$ est bijective.

    Vérifions que $u$ est injective.
    \begin{itemize}
        \item Soit $s \in \ker(u)$.
        \item On sait que, comme $s \in E = F \oplus S'$, il existe un unique couple $(f,s') \in F \times S'$ tel que $s = f + s'$.
        \item Mais $s \in \ker(u)$, donc $u(s) = p(s) = s' = 0$. Donc $s = f + 0 = f \in F$.
        \item Mais $s \in S$ et $F \cap S = \{0\}$ parce que $S$ est un supplémentaire de $F$. Donc $s = 0$.
        \item Ainsi, $\ker(u) = \{0\}$ et donc $u$ est injective.
    \end{itemize}
    Vérifions que $u$ est surjective.
    \begin{itemize}
        \item Soit $s' \in S'$.
        \item Comme $E = F \oplus S$, il existe un unique couple $(f,s) \in F \times S$ tel que $s' = f + s$.
        \item Donc $p(s') = p(f + s) = s$ (car $p$ est la projection sur $S'$ parallèlement à $S$).
        \item Ainsi, pour tout $s' \in S'$, il existe un $s \in S$ tel que $u(s) = p(s) = s'$. Donc $u$ est surjective.
    \end{itemize}
    On a donc montré que $u$ est bijective et linéaire, donc c'est un isomorphisme entre $S$ et $S'$.
\end{proof}

\begin{definition}[Codimension]
    $F$ est dit de \textbf{codimension finie} dans $E$ s'il admet un supplémentaire $S$ de dimension finie. Dans ce cas, on définit la \textbf{codimension} de $F$ dans $E$ par :
    $$\codim_E(F) = \dim(S)$$
\end{definition}

\begin{methodbox}
    Pour construire un supplémentaire, il suffit généralement d'utiliser le théorème de la base incomplète. Dans un espace de dimension infinie, on utilise l'axiome de Zorn (pas étudié) pour construire un supplémentaire.
\end{methodbox}

\begin{example}
    Dans l'espace des polynômes $E = \R[X]$, on pose $P \in E\setminus\{0_E\}$ et $F = P\R[X]$, qui est bien un sous-espace vectoriel de $E$ de dimension infinie ($\dim F = \infty$).

    Cherchons si $F$ admet un supplémentaire de dimension finie. On cherche donc un sous-espace vectoriel $S$ qui vérifie la condition suivante :
    $$\R[X] = P\R[X] \oplus S$$

    Cette écriture rappelle la division euclidienne des polynômes. En posant $d = \deg(P)$, on peut choisir $S = \R_{d-1}[X]$, qui est de dimension finie $\dim S = d$. En effet, pour tout $Q \in \R[X]$, on peut écrire $Q = P R + S$ avec $R \in \R[X]$ et $S \in \R_{d-1}[X]$ (division euclidienne). De plus, l'intersection $P\R[X] \cap \R_{d-1}[X] = \{0\}$ car un polynôme de degré inférieur à $d$ ne peut pas être multiple de $P$ (qui est de degré $d$).
    
    Ainsi, $\R[X] = P\R[X] \oplus \R_{d-1}[X]$ et donc $\codim_{\R[X]}(P\R[X]) = d$.
\end{example}

\begin{example}
    Dans l'espace des matrices carrées de taille $n \in \N^*$ $E = \mathcal M_n(\R)$, on pose $H = \{X \in E \mid \Tr(X) = 0\}$. C'est bien un sous-espace vectoriel de $E$ de dimension finie, vérifions que sa codimension est de $1$.

    Pour cela, il suffit simplement de calculer la dimension de $H$. On peut utiliser les caractéristiques de l'application trace :
    $$\Tr : \begin{matrix} E & \longrightarrow & \R \\ X & \longmapsto & \Tr(X) \end{matrix}$$
    qui est une forme linéaire. Le sous-espace $H$ est donc, par définition, le noyau de cette forme linéaire. Comme $\Tr$ est non-nulle, $H$ est un hyperplan de $E$ et donc $\dim H = \dim E - 1 = n^2 - 1$. Ainsi, $\codim_E(H) = 1$.

    Il est aussi possible d'aborder ce calcul de dimension via théorème de rang.
\end{example}

\begin{remark}
    La codimension de $F$ ne dépend pas du supplémentaire $S$ choisi car n'importe quel supplémentaire est isomorphe à $S$, donc leurs dimensions sont égales.
\end{remark}

\chapter{Hyperplans}

\begin{definition}[Hyperplan en dimension quelconque]
    On dit qu'un sous-espace vectoriel $H$ d'un $\Kset$-ev. $E$ est un hyperplan si et seulement si $\codim_E(H) = 1$, c'est-à-dire qu'il admet un supplémentaire de dimension $1$.
\end{definition}

\begin{proposition}[Formule du rang en dimension quelconque]
    Soit $u \in \mathcal L(E,F)$ une application linéaire. Les propriétés suivantes sont équivalentes :
    \begin{enumerate}
        \item $\rang(u)$ est de dimension finie ($\dim(\Imf(u)) < \infty$).
        \item $\ker(u)$ est de codimension finie dans $E$ ($\codim_E(\ker(u)) < \infty$).
    \end{enumerate}
    Le noyau admet alors un supplémentaire de dimension finie si et seulement si l'image est de dimension finie. De plus, on a la formule du rang :
    $$\dim(\Imf(u)) = \codim_E(\ker(u))$$
\end{proposition}

\begin{proof}
    Montrons que les deux propriétés sont équivalentes. \textbf{Montrons d'abord que (1) $\implies$ (2).}

    On pose $r = \rang(u)$, donc $\exists(e_1,\dots,e_r)\in E$ tel que $(u(e_1),\dots,u(e_r))$ est une base de $\Imf(u)$. Montrons que la codimension du noyau est finie, c'est-à-dire qu'il admet un supplémentaire de dimension finie.

    En dimension finie, le résultat est immédiat : la dimension du supplémentaire est égale à la dimension de l'image (rang $r$). En dimension infinie, on peut construire un supplémentaire de $\ker(u)$ de dimension finie en posant :
    $$S = \Vect(e_1,\dots,e_r)$$
    ...qui est clairement de dimension finie.

    Montrons que $\ker(u) \cap S = \{0\}$. Soit $x \in \ker(u) \cap S$. Alors $x = \sum_{i=1}^r \lambda_i e_i$ pour certains $\lambda_i \in \Kset$. Comme $x \in \ker(u)$, on a $u(x) = 0$, donc :
    $$0 = u(x) = u\left(\sum_{i=1}^r \lambda_i e_i\right) = \sum_{i=1}^r \lambda_i u(e_i)$$
    Mais $(u(e_1),\dots,u(e_r))$ est une base (donc famille libre) de $\Imf(u)$, donc les scalaires $\lambda_i$ sont tous nuls. Ainsi, $x = 0$ et donc $\ker(u) \cap S = \{0\}$.
    
    Montrons que $E = \ker(u) + S$. Écrivons $x \in E$ sous la forme $k + s$ avec $k \in \ker(u)$ et $s \in S$. On procède par analyse-synthèse :
    \begin{itemize}
        \item \textbf{Analyse :} Soit $x \in E$. $x = k+s =  k + \sum_{i=1}^r\lambda_i e_i \implies u(x) = \sum_{i=1}^r\lambda_i u(e_i)$. Comme $(u(e_1),\dots,u(e_r))$ est une base de $\Imf(u)$, il existe des scalaires $\lambda_i$ tels que $u(x) = \sum_{i=1}^r\lambda_i u(e_i)$. Ensuite, on prend $k = x - \sum_{i=1}^r\lambda_i e_i$ et on a $u(k) = u(x) - \sum_{i=1}^r\lambda_i u(e_i) = 0$, donc $k \in \ker(u)$.
        \item \textbf{Synthèse :} Soit $x \in E$, donc $u(x) \in \Imf(u) = \Vect(u(e_1),\dots,u(e_r))$. Donc il existe des scalaires $\lambda_i$ tels que $u(x) = \sum_{i=1}^r\lambda_i u(e_i)$. Par linéarité de $u$, on a $u(x - \sum_{i=1}^r\lambda_i e_i) = u(x) - \sum_{i=1}^r\lambda_i u(e_i) = 0$, donc $x - \sum_{i=1}^r\lambda_i e_i \in \ker(u)$. Posons $k = x - \sum_{i=1}^r\lambda_i e_i \in \ker(u)$ et $s = \sum_{i=1}^r\lambda_i e_i \in S$. Alors $x = k + s$ avec $k \in \ker(u)$ et $s \in S$.
    \end{itemize}


    On conclut alors que $E = \ker(u) \oplus S$, ce qui prouve que $\rang(u) < \infty \implies \codim_E(\ker(u)) < \infty$.

    Montrons maintenant que $\rang(u) < \infty \implies \codim_E(\ker(u)) = \rang(u)$. On sait que $S = \Vect(e_1,\dots,e_r)$ est un supplémentaire de $\ker(u)$. On sait que la famille $(u(e_1),\dots,u(e_r))$ est une base de $\Imf(u)$, donc elle est libre, et on a $\sum_{i=1}^r \lambda u(e_i) = 0 \implies \lambda_i = 0$ pour tout $i$. Donc $\sum_{i=1}^r \lambda e_i \in S \cap \ker(u) = \{0\}$, donc $\lambda_i = 0$ pour tout $i$. Ainsi, $(e_1,\dots,e_r)$ est une famille libre de $S$, donc c'est une base de $S$ et donc $\dim(S) = r = \rang(u)$. Ainsi, $\codim_E(\ker(u)) = \dim(S) = \rang(u)$.

    \textbf{Montrons que (2) $\implies$ (1).} On suppose que $\ker(u)$ est de codimension finie, donc il existe un supplémentaire $S$ de dimension finie tel que $E = \ker(u) \oplus S$. Soit l'application suivante :
    $$f : \begin{matrix} S & \longrightarrow & \Imf(u) \\ s & \longmapsto & u(s) \end{matrix}$$
    Montrons que $f$ est un isomorphisme.
    
    On sait que $f$ est linéaire car $u$ l'est, et donc $\ker(f) = \ker(u) \cap S = \{0\}$, donc $f$ est injective.
    
    Montrons que $f$ est surjective. Soit $y \in \Imf(u)$, donc il existe $x \in E$ tel que $y = u(x)$. Comme $E = \ker(u) \oplus S$, on peut écrire $x = k + s$ avec $k \in \ker(u)$ et $s \in S$. Alors $y = u(x) = u(k + s) = u(k) + u(s) = 0 + u(s) = u(s)$. Ainsi, pour tout $y \in \Imf(u)$, il existe un $s \in S$ tel que $f(s) = y$, donc $f$ est surjective. Ainsi, $f$ est un isomorphisme entre $S$ et $\Imf(u)$, donc $\dim(\Imf(u)) = \dim(S) < \infty$. Ainsi, $\codim_E(\ker(u)) < \infty \implies \rang(u) < \infty$.
\end{proof}

\begin{remark}
    Si la dimension de $E$ est finie, alors $\ker(u)$ est de codimension finie, car $\ker(u)$ est un sous-espace vectoriel de $E$ et donc $\codim_E(\ker(u)) = \dim(E) - \dim(\ker(u)) < \infty$. La formule du rang devient alors :
    $$\dim(\Imf(u)) = \dim(E) - \dim(\ker(u))$$
    Il s'agit de la formule du rang classique.
\end{remark}

\begin{methodbox}
    Pour construire un supplémentaire du noyau, il suffit de prendre les antécédents d'une base $u(e_i)$ de l'image (les $e_i$ sont les antécédents) et construire le sous-espace vectoriel engendré par ces antécédents $\Vect((e_i))$.
\end{methodbox}

\begin{example}
    Posons $F = \{P \in \R[X] \tq P(1) = \int_0^1 P(x) \mathrm dx\}$. Montrons que $\codim F < \infty$ et donner un supplémentaire de $F$ dans $\R[X]$.

    Construisons une application dont $F$ sera le noyau.
    $$u : \begin{matrix} \R[X] & \longrightarrow & \R^2 \\ P & \longmapsto & (P(1),\int_0^1 P(x) \mathrm dx) \end{matrix}$$
    On a $F = \ker(u)$, donc il suffit de montrer que $\rang(u) < \infty$. On sait que $\R^2$ est de dimension finie, donc $\rang(u) \leq 2 < \infty$. Ainsi, $\codim F = \rang(u) < \infty$.

    Il est facile de montrer que $u$ est linéaire et donc que $F$ est un sous-espace vectoriel de $\R[X]$.

    Pour construire un supplémentaire de $F$, il suffit de trouver une base de l'image de $u$. On peut prendre les polynômes $1$ et $X$ :
    $$u(1) = (1,1) \et u(X) = (1,\frac12)$$
    On peut vérifier que $(u(1),u(X))$ est une famille libre de $\R^2$ car $\lambda_1 u(1) + \lambda_2 u(X) = 0 \implies (\lambda_1 + \lambda_2, \lambda_1 + \frac12 \lambda_2) = (0,0) \implies \lambda_1 = 0 \et \lambda_2 = 0$. Ainsi, $(u(1),u(X))$ est une base de $\Imf(u)$ et donc un supplémentaire de $F$ est donné par :
    $$S = \Vect(1,X)$$
\end{example}

\section{Interpolation polynômiale}

\begin{application}[Interpolation polynômiale -- Surjectivité]
    Soit $(x_0, x_1, \dots, x_n) \in \Kset^{n+1}$ différents deux à deux. On définit l'application suivante :
    $$u : \begin{matrix} \Kset[X] & \longrightarrow & \Kset^{n+1} \\ P & \longmapsto & (P(x_0), P(x_1), \dots, P(x_n)) \end{matrix}$$

    Montrons qu'elle est surjective.

    On peut facilement montrer que $u$ est linéaire, ce qui nous permet de calculer son rang en utilisant les formules du rang précédentes.

    On sait que $\Imf(u) \subset \Kset^{n+1}$, donc $\rang(u) \leq n+1$. De plus $\rang(u) = \codim\ker(u)$.
    
    Il suffit donc que montrer que $\codim\ker(u) = n+1$.

    Soit $P \in \ker(u)$, alors $P(x_i) = 0$ pour tout $i = 0, 1, \dots, n$.

    \begin{align*}
        \forall i \in \{0, 1, \dots, n\}, P(x_i) = 0 &\implies \prod_{i=0}^n (X - x_i) \text{ divise } P \text{ sachant que les } x_i \text{ sont deux à deux distincts.}\\
        &\implies \ker(u) = \prod_{i=0}^n (X - x_i) \Kset[X] \\
        &\implies \codim\ker(u) = n+1 \text{car il admet} \Kset_n[X] \text{ comme supplémentaire par DE.}
    \end{align*}

    On en déduit que $\rang(u) = \codim\ker(u) = n+1$ et donc $\Imf(u) = \Kset^{n+1}$, ce qui prouve que $u$ est surjective.
\end{application}

\begin{application}[Polynôme d'interpolation]
    Si on définit l'application $u$ comme dans l'application précédente mais avec ensemble de départ $\Kset_n[X]$ au lieu de $\Kset[X]$, elle devient un isomorphisme.
    $$u : \begin{matrix} \Kset_n[X] & \longrightarrow & \Kset^{n+1} \\ P & \longmapsto & (P(x_0), P(x_1), \dots, P(x_n)) \end{matrix}$$
    En effet, on avait trouvé $\codim\ker(u) = n+1$ car $\Kset[X] = \ker(u) \oplus \Kset_n[X]$.
\end{application}

\begin{proposition}
    Si $x_i \in \Kset$ sont deux à deux distincts, alors :
    $$\exists! P \in \Kset_n[X] \tq \forall i \in \{0, 1, \dots, n\}, P(x_i) = y_i$$
    P s'appelle le polynôme d'interpolation de Lagrange associé aux points $(x_i, y_i)$.
\end{proposition}

On peut donner une expression explicite de $P$. On pose $y = (y_0, \dots, y_n) = \sum_{i=0}^n y_i e_i \in \Kset^{n+1}$, où $e_i$ est la base canonique de $\Kset^{n+1}$.

Il existe un unique $L_i$ dans $\Kset_n[X]$ tel que $e_i = f(L_i)$ avec $f : \Kset_n[X] \to \Kset^{n+1}$ l'application d'interpolation (qu'on avait notée $u$ précédemment). On a donc $P = \sum_{i=0}^n y_i L_i$.

En effet :
\begin{align*}
    e_i = f(L_i) &\iff L_i(x_j) = \delta_{ij} \\
    &\iff \begin{cases}
    \prod_{j=0, j \ne i}^n (X - x_j) \text{ divise } L_i \\
    L_i(x_i) = 1
    \end{cases} \\
    &\iff \begin{cases}
    L_i = c_i\prod_{j=0, j \ne i}^n (X - x_j) \scq L_i \in \Kset_n[X] \\
    L_i(x_i) = 1 \donc c_i = \frac{1}{\prod_{j=0, j \ne i}^n (x_i - x_j)}
    \end{cases}\\
    &\iff L_i = \frac{\prod_{j=0, j \ne i}^n (X - x_j)}{\prod_{j=0, j \ne i}^n (x_i - x_j)}\\
    &\iff \boxed{P = \sum_{i=0}^n y_i L_i = \sum_{i=0}^n y_i \frac{\prod_{j=0, j \ne i}^n (X - x_j)}{\prod_{j=0, j \ne i}^n (x_i - x_j)}}
\end{align*}

$P$ s'appelle alors le polynôme d'interpolation de Lagrange associé aux points $(x_i, y_i)$.

\section{Propriétés sur les sous-espaces vectoriels}

\begin{proposition}
    Si $F$ et $G$ sont des sous-espaces vectoriels de $E$ tels que $F \subset G$, alors :
    \begin{itemize}
        \item En dimension finie : $\dim F \le \dim G$ avec égalité si et seulement si $F = G$.
        \item En dimension infinie : si $\codim F$ est de dimension finie, alors $\codim G$ est de dimension finie et $\codim G \leq \codim F$. Il y a égalité si et seulement si $F = G$.
    \end{itemize}
\end{proposition}

\begin{proof}
    En effet, comme $F \subset G$ et que la codimension de $F$ est finie, alors $E = F \oplus S$.
    $$\codim F < \infty \implies E = F \oplus S \tq \dim S < \infty$$

    $S \cap G$ est un sous-espace vectoriel de $S$. Il existe donc un sous-espace vectoriel $T$ de $S$ tel que $S = (S \cap G) \oplus T$. On a donc :
    $$E = F \oplus S = F \oplus (S \cap G) \oplus T$$
    Comme $F \subset G$, on a $F \oplus (S \cap G) \subset G$.
    $$E \subset G + T \implies E = G + T$$
    On vérifie que $G \cap T = \{0\}$. On a $G \cap T \subset G \cap S \cap T$ (car $T \subset S$). Mais $S = (S \cap G) \oplus T$, donc $S \cap T = \{0\}$. Ainsi, $G \cap T = \{0\}$ et donc $E = G \oplus T$.

    Donc $\codim G = \dim T \le \dim S = \codim F$. De plus, si $\codim G = \codim F$, alors $\dim T = \dim S$ et donc $S = T$ (car $T \subset S$). Ainsi, $S \cap G = \{0\}$ et donc $F \oplus (S \cap G) = F$. Mais $F \oplus (S \cap G) = G$ car $E = F \oplus (S \cap G) \oplus T = G \oplus T$. Donc $F = G$.
\end{proof}

\begin{methodbox}
    En pratique, la proposition précédente nous permet de montrer une égalité de sous-espaces vectoriels quand un sens de l'inclusion est connu et que l'on peut calculer les dimensions ou codimensions et montrer qu'elles sont égales.
\end{methodbox}

\begin{property}
    Soit $F$ et $G$ deux sous-espaces vectoriels de $E$ tels que $F \subset G$. Si $T$ est un supplémentaire commun à $F$ et $G$, alors $F = G$, i.e.:
    $$E = F \oplus T = G \oplus T \et F \subset G \implies F = G$$
\end{property}

\begin{proof}
    En dimension finie, il suffit d'utiliser $\dim F = \dim E - \dim T$ et $\dim G = \dim E - \dim T$ pour conclure que $F = G$ comme $F \subset G$.

    En dimension quelconque, on procède par double inclusion. Un sens est déjà connu : $F \subset G$. Il suffit de montrer que $G \subset F$. Soit $g \in G$.
    
    On peut encore le ``regarder'' comme élément de $E$, donc il peut s'écrire comme $g = f + t$ avec $f \in F$ et $t \in T$. Comme $g \in G$, on a $t = g - f \in G$. Mais $t \in T$ et $T$ est un supplémentaire de $G$ donc leur intersection $G \cap T = \{0\}$.
    
    Ainsi, $t = 0$ et donc $g = f \in F$. On a donc bien $G \subset F$.

    On en déduit que $F = G$.
\end{proof}

\begin{consequence}[Hyperplan maximal]
    Soient $H$ un hyperplan de $E$ et $G$ un sous-espace vectoriel de $E$ tel que $H \subset G$. Alors $G = H$ ou $G = E$.
    $$H \subset G \subset E \implies G = H \ou G = E$$
\end{consequence}

\begin{proof}
    En dimension infinie, la démonstration est triviale : $G$ ne peut être soit de dimension $\dim E - 1$ (et donc égal à $H$) soit de dimension $\dim E$ (et donc égal à $E$).

    En dimension infinie, on sait que $\codim H = 1$ comme c'est un hyperplan. Comme $H \subset G$, alors $\codim G \le \codim H = 1$. Donc $\codim G = 0$ ou $\codim G = 1$. Si $\codim G = 0$, alors $G = E$. Si $\codim G = 1$, alors $G$ est un hyperplan et donc $G = H$, parce qu'il y aura inclusion avec les mêmes codimensions.
\end{proof}

\begin{remark}
    Si $H_1$ et $H_2$ sont deux hyperplans de $E$ et $H_1 \subset H_2$, alors $H_1 = H_2$.
\end{remark}

\begin{remark}
    Si $H$ est un hyperplan de $E$ et $x_0 \in E\setminus H$, alors $E = H \oplus \Vect(x_0)$.
\end{remark}

\begin{proof}
    En effet, on a $H \subset H \oplus \Kset x_0 \subset E$. Comme $H \ne H \oplus \Kset x_0$ (car $x_0 \notin H$), alors par le corollaire précédent, on a $H \oplus \Kset x_0 = E$.
\end{proof}

\begin{exercise}
    Soit $F$ un sous-espace vectoriel de $E$ tel que $\codim F < \infty$ et soit $G$ un sous-espace vectoriel de $E$ tel que $F \cap G = \{0\}$. Montrer que $G$ est de dimension finie et que $\dim G \leq \codim F$. De plus, montrer que l'égalité a lieu si et seulement si $E = F \oplus G$.
\end{exercise}

\begin{correction}
    On a $F \subset F \oplus G$ (qui est une somme directe parce que l'intersection $F \cap G$ est réduite à 0). On sait aussi que la codimension de $F$ est finie, alors $\codim (F\oplus G) \le \codim F < \infty$. Donc $\exists T$ un sous-espace vectoriel de dimension finie tel que $E = F \oplus (G \oplus T)$. Or $\codim F < \infty$, donc il existe un supplémentaire $S$ de dimension finie tel que $E = F \oplus S$, donc $S \simeq G \oplus T$ (isomorphisme) avec $\dim S < \infty$, ce qui implique que $\dim G < \infty$ et $\dim G \le \dim S = \codim F$.

    Pour montrer que $\dim G = \codim F \iff F \oplus G = E$; dans le sens direct, si $\dim G = \codim F$, alors $\dim S = \dim G + \dim T$ et comme $\dim S = \codim F$, on a $\dim T = 0$, donc $T = \{0\}$ et donc $E = F \oplus G$. Dans le sens réciproque, si $E = F \oplus G$, alors $\codim F = \dim G$.

\chapter{Équations d'hyperplans}

\begin{customenv}{\faIcon{lightbulb}}{Éq. d'un hyp. en dim. finie}
    Soit $H$ un hyperplan de $E$. Dans le cas où $\dim E = n < \infty$, on pose $e = (e_1,\dots,e_n)$ une base de $E$. Alors il existe des scalaires $\alpha_1,\dots,\alpha_n \in \Kset\setminus\{0\}$ tels que :
    $$H = \{x \in E \tq \alpha_1 x_1 + \dots + \alpha_n x_n = 0\} \text{ où } x = \sum_{i=1}^n x_i e_i$$
    On dit que $H$ est l'hyperplan d'équation $\alpha_1 x_1 + \dots + \alpha_n x_n = 0$.
\end{customenv}

\begin{proof}
    On fait une étude selon le sens de l'implication.
    \begin{itemize}
        \item[$\implies$] Dans le sens direct, comme $H$ est un hyperplan, c'est donc le noyau d'une forme linéaire non-nulle
        $$H = \ker\varphi \tq \varphi \in E^* \setminus \{0\}$$
        Posons $x = \sum_{i=1}^n x_i e_i \in H$. Alors $\varphi(x) = \sum_{i=1}^n x_i \varphi(e_i) = 0$. Posons $\alpha_i = \varphi(e_i)$ pour $i=1,\dots,n$. Comme $\varphi \ne 0$, il existe au moins un $i$ tel que $\alpha_i \ne 0$. Donc $H = \{x \in E \tq \alpha_1 x_1 + \dots + \alpha_n x_n = 0\}$.
        \item[$\impliedby$] Dans le sens réciproque, on pose $a = (a_1,\dots,a_n) \in \Kset^n \setminus \{0_{\Kset^n}\}$ tel que $H = \{x \in E \tq a_1 x_1 + \dots + a_n x_n = 0\}$ où $x = \sum_{i=1}^n x_i e_i$. Alors on peut définir une forme linéaire $\varphi : E \to \Kset$ par $\varphi(x) = a_1 x_1 + \dots + a_n x_n$ (et vérifier facilement qu'elle est linéaire). Comme $a \ne 0$, on a $\varphi \ne 0$. Donc $H = \ker\varphi$ est un hyperplan.
    \end{itemize}
\end{proof}

\begin{proposition}
    Soit $\varphi, \psi$ deux formes linéaires non-nulles sur $E$. Alors :
    $$\ker\varphi = \ker\psi \iff \exists \lambda \in \Kset\setminus\{0\} \tq \varphi = \lambda \psi$$
    Cette proposition est vraie quelque soit la dimension de $E$.
\end{proposition}

\begin{proof}
    Dans le sens indirect, la démonstration est immédiate : si $\varphi = \lambda \psi$ avec $\lambda \ne 0$, alors $\ker\varphi = \ker\psi$.

    Dans le sens direct, supposons que $\ker\varphi = \ker\psi$. Donc leurs noyaux forment un hyperplan de $E$ qu'on notera $H$, et comme $\varphi$ est une forme linéaire non-nulle, il existe $x_0 \in E$ tel que $\varphi(x_0) \ne 0$. Comme $H$ est un hyperplan, on a $E = H \oplus \Kset x_0$. Donc pour tout $x \in E$, il existe un unique couple $(h, \lambda) \in H \times \Kset$ tel que $x = h + \lambda x_0$. Alors :
    \begin{align*}
        \varphi(x) &= \varphi(h + \lambda x_0) = \varphi(h) + \lambda \varphi(x_0) = 0 + \lambda \varphi(x_0) = \lambda \varphi(x_0) \\
        \psi(x) &= \psi(h + \lambda x_0) = \psi(h) + \lambda \psi(x_0) = 0 + \lambda \psi(x_0) = \lambda \psi(x_0)
    \end{align*}
    Donc $\varphi(x) = \lambda \varphi(x_0) = \frac{\varphi(x_0)}{\psi(x_0)} \psi(x)$. Posons $\lambda = \frac{\varphi(x_0)}{\psi(x_0)} \ne 0$, alors $\varphi = \lambda \psi$ (sachant que les restrictions de ces formes linéaires à $H$ sont nulles donc égales).
\end{proof}

\begin{customenv}{\faIcon{lightbulb}}{Éq. d'un hyp. en dim. infinie}

\chapter{Intersection finies d'hyperplans}

\begin{proposition}
    \label{proposition:hyperplans_independants}
    Soit $(\varphi_1,\dots,\varphi_r)$ des formes linéaires non-nulles sur $E$. Alors :
    $$u : E \to \Kset^r$$
    $$x \mapsto (\varphi_1(x),\dots,\varphi_r(x))$$
    est une application linéaire surjective si et seulement si la famille $(\varphi_1,\dots,\varphi_r)$ est libre.

    Dans ce cas, $H_i = \ker\varphi_i$ sont dits hyperplans indépendants et on a :
    $$\codim \bigcap_{i=1}^r H_i = r$$

    On dit que des hyperplans sont indépendants si et seulement si les formes linéaires qui les définissent sont indépendantes.
\end{proposition}

\begin{proof}
    Montrons d'abord que $u$ est une application linéaire. On a $u(\lambda x + y) = (\varphi_1(\lambda x + y), \dots, \varphi_r(\lambda x + y)) = (\lambda \varphi_1(x) + \varphi_1(y), \dots, \lambda \varphi_r(x) + \varphi_r(y)) = \lambda u(x) + u(y)$, donc $u$ est bien linéaire.

    Montrons maintenant que $u$ surjective $\iff$ $(\varphi_1,\dots,\varphi_r)$ est libre. On rappelle que $u$ surjective $\iff \Imf(u) = \Kset^r$. Nous allons raisonner par contraposée, en montrant que si la famille $(\varphi_1,\dots,\varphi_r)$ est liée, alors $u$ n'est pas surjective.
    $$(\varphi_1,\dots,\varphi_r) \text{ liée } \implies \exists (\lambda_1,\dots,\lambda_r) \in \Kset^r \setminus \{0_{\Kset^r}\} \tq \sum_{i=1}^r \lambda_i \varphi_i = 0$$

    $(\varphi_i)$ est liée $\iff$ il existe des scalaires $(\lambda_1,\dots,\lambda_r) \in \Kset^r \setminus \{0_{\Kset^r}\}$ tels que $\sum_{i=1}^r \lambda_i \varphi_i = 0$. Alors on a $\sum_{i=1}^r \lambda_i \varphi_i = 0$. On peut traduire cette équation en hyperplan d'équation $\sum_{i=1}^r \lambda_i y_i = 0$ dans $\Kset^r$. Donc l'image de $u$ est contenue dans l'hyperplan d'équation $\sum_{i=1}^r \lambda_i y_i = 0$, donc $u$ n'est pas surjective. Donc $(\varphi_i)$ est liée $\iff$ $\exists H$ un hyperplan de $\Kset^r$ tel que $\forall x \in E, u(x) = (\varphi_1(x), \dots, \varphi_r(x)) \in H$.
    
    Ainsi, $\Imf(u) \subset H$, donc $u$ n'est pas surjective, parce que $\Imf(u) \subset H \iff \dim(\Imf(u)) \le \dim(H) < \dim(\Kset^r)$ (voir \ref{prop:dim_inclusion_hyperplans}, ce n'est pas évident !).
    
    Donc $(\varphi_i)$ est liée $\iff$ $u$ n'est pas surjective. Ainsi, $(\varphi_i)$ libre $\iff$ $u$ surjective. 

    Pour montrer que $\codim \bigcap_{i=1}^r H_i = r$, on pose $H_i = \ker\varphi_i$. Donc $\bigcap_{i=1}^r H_i = \ker(u)$, et par la formule du rang, on a $\codim \ker(u) = \dim(\Imf(u)) = r$, sachant que $u$ est surjective.
\end{proof}

\begin{property}
    \label{prop:dim_inclusion_hyperplans}
    Soit $E$ un espace vectoriel de dimension $r$ et $F$ un sous-espace vectoriel de $E$. Alors :
    $$\dim F \le r-1 \iff \exists G \text{ hyperplan } \tq F \subset G$$
\end{property}

\begin{proof}
    Dans le sens indirect, la démonstration est évidente par propriétés simples de la dimension. Dans le sens direct, la démonstration est plus subtile et repose sur le théorème de la base incomplète.

    On pose $d = \dim F$, donc $(e_1, \dots, e_d)$ est une base de $F$. Comme $d \le r-1$, on peut compléter cette base en une base de $E$ $(e_1, \dots, e_d, e_{d+1}, \dots, e_r)$. On pose alors $G = \Vect(e_1, \dots, e_{r-1})$. Alors $G$ est un hyperplan de $E$ et $F \subset G$.
\end{proof}

\begin{example}
    On pose $F = \{P \in \R[X] \mid P(1) = \int_0^1 P = 0\}$.
    Calculons la codimension de $F$.

    On a $F = \ker(\varphi_1) \cap \ker(\varphi_2)$ avec $\varphi_1(P) = P(1)$ et $\varphi_2(P) = \int_0^1 P$. Montrons que $(\varphi_1, \varphi_2)$ est une famille libre.

En prenant $P = 1$, on obtient $\lambda_1 + \lambda_2 = 0$. En prenant $P = X-1$, on obtient $\frac{\lambda_2}{2} = 0$. On obtient $\lambda_1 = 0$ et $\lambda_2 = 0$. Donc $(\varphi_1, \varphi_2)$ est une famille libre.

\chapter{Dualité}

\begin{definition}[Dual]
    On appelle le dual de $E$, noté $E^*$, l'espace des formes linéaires sur $E$ : $E^* = \mathcal L(E, \Kset)$ (à ne pas confondre avec $E\setminus \{0\}$).
\end{definition}

\begin{proposition}
    Si $\dim E = n < \infty$, alors $\dim E^* = n$.
\end{proposition}

\begin{proof}
    Si $E$, $F$ sont de dimensiosn finies, alors $\dim \mathcal L(E,F) = \dim E \cdot \dim F$. En particulier pour $F = \Kset$, on trouve $\dim E^* = n \times 1 = n$.
\end{proof}

\begin{proposition}[Base duale]
    Soit $(e_1, \dots, e_n)$ une base de $E$. On définit la base duale $e^* = (e_1^*, \dots, e_n^*)$ de $E^*$ par :
    $$e_i^* : \begin{matrix} E & \longrightarrow & \Kset \\ x & \longmapsto & x_i \end{matrix}$$
    avec $x_i$ la $i$-ème coordonnée de $x$ dans la base $(e_1, \dots, e_n)$.
\end{proposition}

\begin{proof}
    Nous allons montrer que $e^* = (e_1^*, \dots, e_n^*)$ est une base de $E^*$. Pour cela, on montre que $\forall i, e_i^* \in E^*$ et que $(e_1^*, \dots, e_n^*)$ est une famille libre de $E^*$.

    Montrons que $e_i^* \in E^*$.

    On a $x = \sum_{i=1}^n x_i e_i$ et $y = \sum_{i=1}^n y_i e_i$. Alors $e_i^*(x) = x_i$ et $e_i^*(y) = y_i$. Soit $\lambda \in \Kset$. Alors :
    \begin{align*}
        e_i^*(\lambda x + y) &= e_i^*\left(\sum_{j=1}^n (\lambda x_j + y_j) e_j\right) \\
        &= \lambda x_i + y_i \\
        &= \lambda e_i^*(x) + e_i^*(y)
    \end{align*}
    Donc $e_i^*$ est linéaire, donc $e_i^* \in E^*$.

    Montrons que $e_i^*$ est une famille libre de $E^*$.
    
    Supposons que $\sum_{i=1}^n \lambda_i e_i^* = 0$. Alors pour tout $x \in E$, on a :
    $$\sum_{i=1}^n \lambda_i e_i^*(x) = 0$$
    En particulier, pour $x = e_j$, on a $e_i^*(e_j) = \delta_{ij}$ (le symbole de Kronecker). Donc :
    $$\sum_{i=1}^n \lambda_i e_i^*(e_j) = \sum_{i=1}^n \lambda_i \delta_{ij} = \lambda_j = 0$$
    Donc $\lambda_j = 0$ pour tout $j = 1, \dots, n$. Ainsi, $(e_1^*, \dots, e_n^*)$ est une famille libre de $E^*$.

    Comme $\dim E^* = n$, on conclut que $(e_1^*, \dots, e_n^*)$ est une base de $E^*$, vu qu'elle contient $n$ éléments.

    \textbf{Méthode alternative : } Il est possible de montrer que $(e_i^*)$ est une base de $E^*$ en montrant que c'est une famille génératrice de $E^*$. On prend alors une forme linéaire quelconque $\varphi \in \mathcal L(E,\Kset)$ et $\forall x \in E$, on a $\varphi(x) = \sum_{i=1}^n x_i \varphi(e_i) = \sum_{i=1}^n \varphi(e_i) e_i^*(x)$. Donc $\varphi = \sum_{i=1}^n \varphi(e_i) e_i^*$, ce qui montre que $(e_i^*)$ est génératrice de $E^*$ vu que cette égalité est valable pour tout $x \in E$. Comme $\dim E^* = n$, on conclut que $(e_i^*)$ est une base de $E^*$. \textbf{Cette méthode nous permet aussi d'obtenir les coordonnées de $\varphi$ dans la base duale : } $\varphi = \sum_{i=1}^n \varphi(e_i) e_i^*$, donc les coordonnées de $\varphi$ dans la base duale sont $(\varphi(e_1), \dots, \varphi(e_n))$.
\end{proof}

\begin{remark}
    Les coordonnées de $\varphi \in \mathcal L(E,\Kset)$ dans la base duale sont données par $\varphi(e_i)$ pour $i=1,\dots,n$.
\end{remark}

\begin{example}
    Dans $E = \R^2$ on pose $\varepsilon_1 = (1,1)$ et $\varepsilon_2 = (1,-1)$. Montrons que $\varepsilon = (\varepsilon_1, \varepsilon_2)$ est une base de $E$ et calculons sa base duale $\varepsilon^*$.

    Le déterminant de la matrice de passage de la base canonique à la base $\varepsilon$ est :
    $$\det\begin{pmatrix}1 & 1 \\ 1 & -1\end{pmatrix} = -2 \ne 0$$
    Donc $\varepsilon$ est une base de $E$. On a $\varepsilon_1^*(\varepsilon_1) = 1$, $\varepsilon_1^*(\varepsilon_2) = 0$, $\varepsilon_2^*(\varepsilon_1) = 0$ et $\varepsilon_2^*(\varepsilon_2) = 1$. Donc la base duale est :
    $$\varepsilon^* = (\varepsilon_1^*, \varepsilon_2^*)$$

    Déterminons alors $\varepsilon_1^*$ et $\varepsilon_2^*$. Soit $x \in E$. On a $x = x_1 \varepsilon_1 + x_2 \varepsilon_2$ avec $x_1, x_2 \in \R$. Alors :
    $$X_1 = \varepsilon_1^*(x) \et X_2 = \varepsilon_2^*(x)$$
    On a donc le système suivant :
    $$(x_1,x_2) = X_1(\varepsilon_1) + X_2(\varepsilon_2)$$
    On peut résoudre ce système pour obtenir $x_1$ et $x_2$ :
    $$x_1 = \frac{X_1 + X_2}{2} \et x_2 = \frac{X_1 - X_2}{2}$$
    Donc on a :
    $$\varepsilon_1^*(x) = \frac{x_1 + x_2}{2} \et \varepsilon_2^*(x) = \frac{x_1 - x_2}{2}$$
\end{example}

    Calculons le déterminant de la matrice des coordonnées de $(\varphi_1, \varphi_2, \varphi_3)$ dans la base duale $(e_0^*, e_1^*, e_2^*)$ :
    $$\det\begin{pmatrix}
        1 & 0 & 1 \\
        1 & 1 & \frac{1}{2} \\
        1 & 2 & \frac{1}{3}
    \end{pmatrix} = 1 \cdot (1 \cdot \frac{1}{3} - 2 \cdot \frac{1}{2}) - 0 + 1 \cdot (1 \cdot 2 - 1 \cdot 1) = 1 \cdot (\frac{1}{3} - 1) + 1 \cdot (2 - 1) = -\frac{2}{3} + 1 = \frac{1}{3} \ne 0$$

    On peut aussi simplement vérifier que la famille est libre, et comme elle est de cardinal 3, égal à la dimension de $E^*$, alors c'est une base de $E^*$.
    
    En effet, supposons que $\lambda_1 \varphi_1 + \lambda_2 \varphi_2 + \lambda_3 \varphi_3 = 0$. Alors pour tout $P \in \R_2[X]$, on a :
    $$\lambda_1 P(1) + \lambda_2 P'(1) + \lambda_3 \int_0^1 P = 0$$
    
    \begin{itemize}
        \item En prenant $P(X) = (X-1)^2$, on a $P(1) = 0$ et $P'(1) = 0$, donc $\lambda_3 \int_0^1 (X-1)^2 dX = 0$. Comme $\int_0^1 (X-1)^2 dX > 0$, on obtient $\lambda_3 = 0$.
        \item En prenant $P(X) = X-1$, on a $P(1) = 0$ et $P'(1) = 1$, donc $\lambda_2 \cdot 1 + \lambda_3 \cdot \int_0^1 (X-1) dX = 0$. Comme $\lambda_3 = 0$, on obtient $\lambda_2 = 0$.
        \item Enfin, en prenant $P(X) = 1$, on a $P(1) = 1$ et $P'(1) = 0$, donc $\lambda_1 \cdot 1 + \lambda_2 \cdot 0 + \lambda_3 \cdot \int_0^1 1 dX = 0$. Comme $\lambda_2 = \lambda_3 = 0$, on obtient $\lambda_1 = 0$.
        \item Donc $(\varphi_1, \varphi_2, \varphi_3)$ est une famille libre.
    \end{itemize}

    Calculons la base préduale. On pose :
    $$u : \begin{matrix} E & \longrightarrow & \R^3 \\ P & \longmapsto & (\varphi_1(P), \varphi_2(P), \varphi_3(P)) \end{matrix}$$
    Cette application est un isomorphisme. On a $\varepsilon_i^* = u(c_i)$ avec $c = (c_i)$ la base canonique de $\R^3$. On cherche à résoudre l'équation $y = u(P)$ pour $P \in E$ et $y \in \R^3$. On a :
    $$\begin{cases}
        y_1 = P(1) \\
        y_2 = P'(1) \\
        y_3 = \int_0^1 P
    \end{cases}$$
    On pose $P(X) = a_0 + a_1 X + a_2 X^2$. Alors on a :
    $$\begin{cases}
        y_1 = a_0 + a_1 + a_2 \\
        y_2 = a_1 + 2a_2 \\
        y_3 = a_0 + \frac{a_1}{2} + \frac{a_2}{3}
    \end{cases}$$
    On résout ce système pour $a_0, a_1, a_2$ en fonction de $y_1, y_2, y_3$. On obtient :
    $$\begin{cases}
        a_0 = -2y_1 +\frac12 y_2 + 6y_3 \\
        a_1 = 6y_1 -2 y_2 - 6y_3 \\
        a_2 = -3y_1 + \frac32 y_2 + 3y_3
    \end{cases}$$
    On obtient donc :
    $$u^{-1}(y) = (-2y_1 +\frac12 y_2 + 6y_3) + (6y_1 -2 y_2 - 6y_3)X + (-3y_1 + \frac32 y_2 + 3y_3)X^2$$
    Donc la base préduale est :
    $$\implies \varepsilon = (\varepsilon_1, \varepsilon_2, \varepsilon_3) \avec \begin{cases} \varepsilon_1 = u^{-1}(1,0,0) \\ \varepsilon_2 = u^{-1}(0,1,0) \\ \varepsilon_3 = u^{-1}(0,0,1) \end{cases}$$
    Après calculs, on trouve :
    $$\begin{cases}
        \varepsilon_1 = -2 + 6X - 3X^2 \\
        \varepsilon_2 = \frac12 - 2X + \frac32 X^2 \\
        \varepsilon_3 = 3 - 6X + 3X^2
    \end{cases}$$

    On peut aussi déterminer la base préduale en évaluant l'égalité suivante en des polynômes particuliers.

    En effet, $\forall P \in \R_2[X]$, on a :
    $$P = \varphi_1(P) \varepsilon_1 + \varphi_2(P) \varepsilon_2 + \varphi_3(P) \varepsilon_3$$
    Donc $P = P(1) \varepsilon_1 + P'(1) \varepsilon_2 + \int_0^1 P \varepsilon_3$.
    
    On pose donc $P(X) = (X-1)^2$. Alors $P(1) = 0$, $P'(1) = 0$ et $\int_0^1 P = \frac{1}{3}$. Donc on obtient :
    $$\frac{1}{3} \varepsilon_3 = (X-1)^2 \implies \varepsilon_3 = 3(X-1)^2 = 3X^2 - 6X + 3$$

    On pose ensuite $P(X) = X-1$. Alors $P(1) = 0$, $P'(1) = 1$ et $\int_0^1 P = -\frac{1}{2}$. Donc on obtient :
    $$\varepsilon_2 - \frac{1}{2} \varepsilon_3 = X-1 \implies \varepsilon_2 = X-1 + \frac{1}{2} \varepsilon_3 = X-1 + \frac{1}{2}(3X^2 - 6X + 3) = \frac{3}{2}X^2 - 2X + \frac{1}{2}$$

    Enfin, on pose $P(X) = 1$. Alors $P(1) = 1$, $P'(1) = 0$ et $\int_0^1 P = 1$. Donc on obtient :
    $$\varepsilon_1 + \varepsilon_3 = 1 \implies \varepsilon_1 = 1 - \varepsilon_3 = 1 - (3X^2 - 6X + 3) = -3X^2 + 6X - 2$$

    Et donc on retrouve bien la base préduale :
    $$\begin{cases}
        \varepsilon_1 = -3X^2 + 6X - 2 \\
        \varepsilon_2 = \frac{3}{2}X^2 - 2X + \frac{1}{2} \\
        \varepsilon_3 = 3X^2 - 6X + 3
    \end{cases}$$
\end{example}

\begin{proposition}
    \label{prop:equivalence_base_duale}
    Soit $E$ un espace de dimension finie $n$ et $e = (e_1,\dots,e_n)$ une \textit{famille} quelconque de $E$, avec $\varphi = (\varphi_1,\dots,\varphi_n)$ une famille quelconque de $E^*$. Alors les propriétés suivantes sont équivalentes :
    \begin{enumerate}
        \item $e$ est une base de $E$ et $\varphi$ est la base duale de $e$ (c'est-à-dire $\varphi = e^*$).
        \item $\forall i,j \in \{1,\dots,n\}, \varphi_i(e_j) = \delta_{ij}$.
        \item $\forall x \in E, x= \sum_{i=1}^n \varphi_i(x) e_i$.
        \item $\forall f \in E^*, f = \sum_{i=1}^n f(e_i) \varphi_i$.
    \end{enumerate}
\end{proposition}

\begin{proof}
    Montrons que (1) $\implies$ (2). Si $e$ est une base de $E$ et $\varphi$ est la base duale de $e$, alors par définition de la base duale, on a $\varphi_i(e_j) = \delta_{ij}$ pour tout $i,j \in \{1,\dots,n\}$.

    Montrons que (2) $\implies$ (1).
    \begin{itemize}
        \item On a $\sum_{i=1}^n \lambda_i e_i = 0 \implies \forall j, \varphi_j\left(\sum_{i=1}^n \lambda_i e_i\right) = \sum_{i=1}^n \lambda_i \varphi_j(e_i) = \lambda_j = 0$. Donc $\lambda_i = 0$ pour tout $i$, donc $e$ est libre. Comme $e$ a $n$ éléments et que $\dim E = n$, alors $e$ est une base de $E$.
        \item Pour montrer que $e_* = \varphi$, il suffit de montrer que ces deux familles coïncident sur la base $e$. On a $\forall i,j, \varphi_i(e_j) = \delta_{ij} = e_i^*(e_j)$. Donc $\varphi_i$ et $e_i^*$ coïncident sur la base $e$, donc $\varphi = e^*$.
    \end{itemize}

    Montrons que (1) $\implies$ (3). Si $e$ est une base de $E$ et $\varphi$ est la base duale de $e$, alors pour tout $x \in E$, on peut écrire $x = \sum_{i=1}^n x_i e_i$. Alors $\varphi_j(x) = \varphi_j\left(\sum_{i=1}^n x_i e_i\right) = \sum_{i=1}^n x_i \varphi_j(e_i) = x_j$. Donc $x = \sum_{i=1}^n \varphi_i(x) e_i$.

    Montrons que (3) $\implies$ (1).
    \begin{itemize}
        \item Si $\forall x \in E, x= \sum_{i=1}^n \varphi_i(x) e_i$, alors la famille $e$ est génératrice de $E$. Comme $e$ a $n$ éléments et que $\dim E = n$, alors $e$ est une base de $E$.
        \item Comme les coordonnées de $x$ dans la base $e$ sont uniques, alors $\varphi_i(x)$ sont les coordonnées de $x$ dans la base $e$, donc $\varphi = e^*$.
    \end{itemize}

    Montrons que (1) $\implies$ (4). Si $e$ est une base de $E$ et $\varphi$ est la base duale de $e$, alors pour tout $f \in E^*$, il existe des scalaires $\lambda_i$ tels que $f = \sum_{i=1}^n \lambda_i \varphi_i$. En évaluant cette égalité en $e_j$, on obtient $f(e_j) = \sum_{i=1}^n \lambda_i \varphi_i(e_j) = \lambda_j$. Donc $\lambda_j = f(e_j)$ et donc $f = \sum_{i=1}^n f(e_i) \varphi_i$.

    Montrons que (4) $\implies$ (1).
    \begin{itemize}
        \item On sait que $\forall f \in E^*, f = \sum_{i=1}^n f(e_i) \varphi_i$. En particulier, pour $f = \varphi_j$, on a $\varphi_j = \sum_{i=1}^n \varphi_j(e_i) \varphi_i$.
        \item En évaluant cette égalité en $e_k$, on obtient $\varphi_j(e_k) = \sum_{i=1}^n \varphi_j(e_i) \varphi_i(e_k)$.
        \item Comme $\varphi_i(e_k)$ est le coefficient de $e_k$ dans la base duale, on obtient que $\varphi_j(e_k) = \delta_{jk}$.
        \item Donc (2) est vérifiée, et par ce qui précède, on conclut que (1) est vérifiée.
    \end{itemize}
\end{proof}

\begin{example}
    Dans $E = \R_n[X]$, on pose $(x_i)$ une famille de $n+1$ réels différents deux à deux. On définit :
    $$\varphi_i : \begin{matrix} \R_n[X] & \longrightarrow & \R \\ P & \longmapsto & P(x_i) \end{matrix}$$
    On note que $\varphi_i$ est un élément de $E^*$ pour chaque $i$.

    On peut utiliser le polynôme d'interpolation de Lagrange pour montrer que $(\varphi_0, \dots, \varphi_n)$ est une base de $E^*$ ; On cherche une famille $(e_i)$ tels que $\varphi_i(e_j) = \delta_{ij}$, c'est-à-dire que $e_j(x_i) = \delta_{ij}$, ce qui rappelle $L_i(x_j) = \delta_{ij}$, le polynôme de Lagrange associé à $x_i$. On peut donc poser $e_i = L_i$ pour $i=0,\dots,n$. On a bien $\varphi_i(e_j) = e_j(x_i) = L_j(x_i) = \delta_{ij}$.

    Donc $(\varphi_0, \dots, \varphi_n)$ est une famille génératrice de $E^*$. Comme elle contient $n+1$ éléments et que $\dim E^* = n+1$, on conclut que $(\varphi_0, \dots, \varphi_n)$ est une base de $E^*$. Sa base préduale est la base de Lagrange $(L_0, \dots, L_n)$.
\end{example}

\begin{example}
    Dans $E = \R_n[X]$, on pose $a \in R$ et on définit :
    $$\varphi_i : \begin{matrix} \R_n[X] & \longrightarrow & \R \\ P & \longmapsto & P^{(i)}(a) \end{matrix}$$
    Montrons que $(\varphi_0, \dots, \varphi_n)$ est une base de $E^*$ et calculons sa base préduale.

    Ca nous rappelle immédiatement le théorème de Taylor. En effet, pour tout $P \in \R_n[X]$, on a :
    $$P(X) = \sum_{i=0}^n \frac{P^{(i)}(a)}{i!} (X-a)^i$$
    Donc on peut poser $e_i = \frac{1}{i!} (X - a)^i$ pour $i=0,\dots,n$.
    
    On sait aussi que $\card{(\varphi_i)_{i \in \{0,\dots,n\}}} = n+1 = \dim(E^*)$.

    Alors, par l'équivalence (1) $\iff$ (3) de la proposition précédente, on conclut que $e = (e_0, \dots, e_n)$ est une base de $E$ et que $\varphi = (\varphi_0, \dots, \varphi_n)$ est la base duale de $e$. Donc $(\varphi_0, \dots, \varphi_n)$ est une base de $E^*$ et sa base préduale est $(e_0, \dots, e_n)$.
\end{example}

\begin{example}

\section{Caractérisation matricielle d'une base et sa base duale}

\begin{proposition}
    Soit $e = (e_1, \dots, e_n)$ une base de $E$, $\varphi = (\varphi_1, \dots, \varphi_n)$ une famille de $E^*$ et $\epsilon = (\epsilon_1, \dots, \epsilon_n)$ la base duale de $e$. On pose aussi :
    $$\begin{cases}
    A = \Mat_e(\epsilon_1, \dots, \epsilon_n) \\
    B = \Mat_{e^*}(\varphi_1, \dots, \varphi_n)
    \end{cases}$$

    Les propriétés suivantes sont équivalentes :
    \begin{enumerate}
        \item $\epsilon$ est une base de $E$ et $\varphi = \epsilon^*$.
        \item $^tAB = I_n$.
    \end{enumerate} 
\end{proposition}

\begin{proof}
    En utilisant la proposition \ref{prop:equivalence_base_duale}, on sait que $\epsilon$ est une base de $E$ et $\varphi = \epsilon^*$ si et seulement si $\forall i,j, \varphi_i(\epsilon_j) = \delta_{ij}$. Il suffit de montrer que cela équivaut à dire que $^tAB = I_n$.

    On sait que $\epsilon_j = \sum_{k=1}^n A_{kj} e_k$ et $\varphi_i = \sum_{l=1}^n B_{li} e_l^*$. Alors :
    \begin{align*}
        \forall i,j, \varphi_i(\epsilon_j) = \delta_{ij} &\iff \forall i,j, \varphi_i\left(\sum_{k=1}^n A_{kj} e_k\right) = \delta_{ij} \\
        &\iff \forall i,j, \sum_{k=1}^n A_{kj} \varphi_i(e_k) = \delta_{ij} \text{ par linéarité de } \varphi_i
    \end{align*}

    On peut considérer que $\varphi_i$ retourne la $i$-ème colonne de la matrice $B$. On peut donc exprimer $\varphi_i = \sum_{l=1}^n B_{li} e_l^*$. Alors :
    \begin{align*}
        \forall i,j, \sum_{k=1}^n A_{kj} \varphi_i(e_k) = \delta_{ij} &\iff \forall i,j, \sum_{k=1}^n A_{kj} \left(\sum_{l=1}^n B_{li} e_l^*(e_k)\right) = \delta_{ij} \text{ sachant que } e_l^*(e_k) = \delta_{lk} \\
        &\iff \forall i,j, \sum_{k=1}^n A_{kj} B_{ki} = \delta_{ij}
    \end{align*}
    Si on veut écrire la dernière expression sous forme de produit matriciel, il faut remarquer que $A_{kj}$ peut faire référence à la $j$-ème ligne et $k$-ème colonne de la transposée de $A$, c'est-à-dire $^tA_{ik}$. Donc on peut réécrire la dernière expression comme suit :
    $$\forall i,j, \sum_{k=1}^n {^tA}_{jk} B_{ki} = \delta_{ij}$$
    $$\iff \forall i,j, (^tAB)_{ji} = \delta_{ij}$$
    $$\iff \boxed{^tAB = I_n}$$
\end{proof}

\chapter{Orthogonalité au sens de la dualité}

\begin{rappel}
    Le produit scalaire $\langle x, y \rangle$ est une forme bilinéaire symétrique définie positive.

    On dit que $x \perp y$ si $\langle x, y \rangle = 0$.

    Pour parler d'orthogonalité, il suffit d'avoir une forme bilinéaire.
\end{rappel}

\begin{definition}[Orthogonalité au sens de la dualité]
    On définit l'application suivante :
    $$B : \begin{matrix}
        E^* \times E & \longrightarrow & \Kset \\ (\varphi, x) & \longmapsto & \langle \varphi, x \rangle = \varphi(x)
    \end{matrix}$$
    ... une forme bilinéaire.

    On dit alors que $\varphi$ et $x$ sont orthogonaux (au sens de la dualité) si $\langle \varphi, x\rangle = \varphi(x) = 0$.

    $$\varphi \perp x \iff \varphi(x) = 0 \iff x \in \ker(\varphi)$$
\end{definition}

\begin{example}
    Dans $E = \R^3$, on pose $\varphi(x) = x_1 + x_2 - 2x_3$. Soit $v = (1,1,1)$. Alors $\varphi(v) = 1 + 1 - 2 \cdot 1 = 0$, donc $\varphi \perp v$.
\end{example}

\begin{definition}[Orthogonal d'une partie]
    Si $A \subset E$, on définit l'orthogonal de $A$ par :
    $$A^\perp = \{\varphi \in E^* \mid \forall x \in A, \varphi(x) = 0\}$$
    C'est l'ensemble des formes linéaires qui s'annulent sur $A$.
    On peut aussi définir l'orthogonal d'une partie $B \subset E^*$ par :
    $$B^\perp = \{x \in E \mid \forall \varphi \in B, \varphi(x) = 0\}$$
\end{definition}

\begin{remark}
    \begin{itemize}
        \item Si $A \subset E^*$, alors $A^\perp \subset E$.
        \item Si $A \subset E$, alors $A^\perp \subset E^*$.
    \end{itemize}
\end{remark}

\begin{proposition}
    Soit $A$ une partie non-vide de $E$ (resp. $E^*$). Alors $A^\perp$ est un sous-espace vectoriel de $E^*$ (resp. $E$).
\end{proposition}

\begin{proof}
    On montrera chaque cas séparément.
    \begin{itemize}
        \item Dans le cas où $A \subset E$, on a $\varphi = 0_{E^*} \in A^\perp$, car $\forall x \in A, 0_{E^*}(x) = 0$. Montrons maintenant la stabilité par combinaisons linéaires. Soient $\varphi,\psi \in A^\perp$ et $\lambda \in \Kset$. Alors $\forall a \in A, \langle \lambda \varphi + \psi, a \rangle = \lambda \langle \varphi, a \rangle + \langle \psi, a \rangle = 0$, donc $\lambda \varphi + \psi \in A^\perp$. Donc $A^\perp$ est un sous-espace vectoriel de $E^*$.
        \item Dans le cas où $A \subset E^*$, on a $0_E \in A^\perp$, il suffit de repartir de la définition de $A^\perp = \{x \in E, \forall \varphi \in A, \varphi(x) = 0\}$ qui est égal à $\bigcap_{\varphi \in A} \ker(\varphi)$. Comme l'intersection de sous-espaces vectoriels est un sous-espace vectoriel, on conclut que $A^\perp$ est un sous-espace vectoriel de $E$. 
    \end{itemize}
\end{proof}

\begin{proposition}
    $$A^\perp = \left[\Vect(A)\right]^\perp$$
    C'est-à-dire que l'orthogonal d'une partie est égal à l'orthogonal de son sous-espace vectoriel engendré.
\end{proposition}

\begin{proof}
    On décompose encore une fois la preuve en deux parties $A \subset E$ et $A \subset E^*$.
    \begin{itemize}
        \item Si $A \subset E$, on a :
        \begin{align*}
        \varphi \in \left(\Vect(A)\right)^\perp &\implies \langle \varphi , v \rangle = 0, \forall v \in \Vect(A) \\
        &\text{en particulier, } \langle \varphi, x \rangle = 0, \forall x \in A \\
        &\implies \varphi \in A^\perp \\
        &\implies \left(\Vect(A)\right)^\perp \subset A^\perp
        \end{align*}
        Dans le sens inverse, on a :
        \begin{align*}
        \varphi \in A^\perp &\implies \langle \varphi, x \rangle = 0, \forall x \in A \\
        &\implies \sum_{i=1}^n \lambda_i \langle \varphi, x_i \rangle = 0, \forall \lambda_i \in \Kset, \forall x_i \in A \\
        &\implies \langle \varphi, v \rangle = 0, \forall v \in \Vect(A) \\
        &\implies \varphi \in \left(\Vect(A)\right)^\perp \end{align*}
        \item Si $A \subset E^*$, on suit la même démarche en utilisant la linéarité de $(\varphi,x) \mapsto \langle \varphi, x \rangle$ à gauche.
    \end{itemize}
\end{proof}

\begin{property}
    Si $A \subset B$, alors $B^\perp \subset A^\perp$.

    i.e.: Tout élément orthogonal à $B$ est orthogonal à $A$.
\end{property}

\begin{proposition}[Orthogonalité et dimension]
    On considère que $E$ est de dimension finie.

    Si $F$ est un sous-espace vectoriel de $E$, alors :
    $$\dim(F) + \dim(F^\perp) = \dim(E)$$

    En posant $n = \dim(E)$, on a $\dim(F^\perp) = n - \dim(F)$.
\end{proposition}

\begin{attention}
    On ne peut plus écrire $F^\perp \oplus F = E$ car $F^\perp$ est un sous-espace de $E^*$ et non de $E$. On ne peut donc pas additionner $F$ et $F^\perp$.
\end{attention}

\begin{proof}
    On décompose la preuve en deux parties $F \subset E$ et $F \subset E^*$.
    \begin{itemize}
        \item SI $F$ un sous-espace vectoriel de $E^*$, alors on pose $(\varphi_1, \dots, \varphi_p)$ une base de $F$. On la complète en une base $(\varphi_1, \dots, \varphi_n)$ de $E^*$. Soit $e$ sa base préduale, donc :
        $$ F = \Vect(\varphi_1, \dots, \varphi_p) \implies F^\perp = \bigcap_{i=1}^p \ker(\varphi_i) = \Vect(e_{p+1}^*, \dots, e_n^*)$$
        $$F^\perp = \left[ \Vect(e_1^*, \dots, e_p^*) \right]^\perp$$
        Donc $\dim(F^\perp) = n - p = n - \dim(F)$.
        \begin{align*}
            x \in F^\perp &\iff \forall i \in \{1, \dots, p\}, e_i^*(x) = 0 \\
            &\iff x \in \bigcap_{i=1}^p \ker(e_i^*) \\
            &\iff \forall i \in \{1, \dots, p\}, x_i = 0 \\
            &\iff x \in \Vect(e_{p+1}^*, \dots, e_n^*) \\
            &\implies \boxed{\dim(F^\perp) = n - p = n - \dim(F)}
        \end{align*}
    \end{itemize}

    Une autre méthode consiste à utiliser une application définie comme suite :
    $$u : \begin{matrix} E^* \longrightarrow F^* \\ \varphi \longmapsto \varphi_{|F} \end{matrix}$$
    Cette application $u$ est linéaire et surjective. En effet, pour tout $f \in F^*$, on peut définir $\varphi \in E^*$ par $\begin{cases} \varphi_{|F} = f \\ \varphi_{|S} = 0 \end{cases}$ où $E = F \oplus S$. Alors $u(\varphi) = f$.
    \begin{align*}
        \dim F &= \dim F^* \\
            &= \rang(u) \\
            &= \dim E^* - \dim(\ker(u)) \\
            &= \dim E^* - \dim(F^\perp) \\
            &= n - \dim(F^\perp) \\
    \end{align*}
    % à revoir: sachant que something (page 7)

    \begin{itemize}
        \item Dans le cas où $F$ est un sous-espace vectoriel de $E$, on pose $(e_1,\dots,e_p)$ une base de $F$ comme suit :
        $$F^\perp = \left[\Vect(e_1, \dots, e_p)\right]^\perp = \left\{e_1, \dots, e_p\right\}^\perp$$
        $$\varphi \in F^\perp \iff \forall i \in \{1, \dots, p\}, \varphi(e_i) = 0$$
        On complète cette base en une base $(e_1, \dots, e_n)$. Alors $\varphi = \sum_{i=1}^n \varphi(e_i)e_i^*$.
        \begin{align*}
            \donc \varphi \in F^\perp &\iff \varphi(e_i) = 0, \forall i \in \{1, \dots, p\} \\
            &\iff \varphi = \sum_{i=p+1}^n \varphi(e_i)e_i^* \\
            &\iff \varphi \in \Vect(e_{p+1}^*, \dots, e_n^*) \\
            &\iff F^\perp = \Vect(e_{p+1}^*, \dots, e_n^*) \\
            &\implies \boxed{\dim(F^\perp) = n - p = n - \dim(F)}
        \end{align*}
    \end{itemize}
\end{proof}

\begin{example}
    Dans $E = \R_3[X]$, on pose :
    $$F = \{P \in \R_3[X] \mid P(1) = \int_0^1 P = 0\}$$
    On cherche à calculer la dimension de $F$.

    On décompose $F$ en intersection d'orthogonaux :
    $$\begin{cases}
        \varphi_1 : P \mapsto P(1) \\
        \varphi_2 : P \mapsto \int_0^1 P
    \end{cases}$$
    Alors $F = \{P \in R_3[X], \langle \varphi_1, P \rangle = 0 \text{ et } \langle \varphi_2, P \rangle = 0\} = \{\varphi_1, \varphi_2\}^\perp = \left[\Vect(\varphi_1, \varphi_2)\right]^\perp$.

\begin{proposition}
    \label{prop:orthogonalite_double}
    Si $F$ est un sous-espace vectoriel de $E$ (resp. $E^*$) alors $\left(F^\perp\right)^\perp = F$.
\end{proposition}

\begin{proof}
    On décompose la preuve en deux parties $F \subset E$ et $F \subset E^*$.
    \begin{itemize}
        \item Dans le cas où $F \subset E$, on pose $x \in F$, alors $\forall \varphi \in F^\perp, \langle \varphi, x \rangle = 0$, donc $x \in (F^\perp)^\perp$. Donc $F \subset (F^\perp)^\perp$. Ainsi, $F \subset (F^\perp)^\perp$.
        \item Dans le cas où $F \subset E^*$, on pose $\varphi \in F$, alors $\forall x \in F^\perp, \langle \varphi, x \rangle = 0$, donc $\varphi \in (F^\perp)^\perp$. Donc $F \subset (F^\perp)^\perp$. Ainsi, $F \subset (F^\perp)^\perp$.
    \end{itemize}
    Dans les deux cas, on a $\dim(F) + \dim(F^\perp) = \dim(E)$ et $\dim(F^\perp) + \dim((F^\perp)^\perp) = \dim(E)$, donc $\dim(F) = \dim((F^\perp)^\perp)$. Comme $F \subset (F^\perp)^\perp$ et que les deux sous-espaces ont la même dimension, on conclut que $F = (F^\perp)^\perp$.
\end{proof}

\begin{proposition}
    \label{prop:orthogonalite_intersection}
    Si $(\varphi_i)_{1 \le i \le r} \in E^*, \varphi \in E^*$, alors $\varphi \in \Vect(\varphi_1, \dots, \varphi_r) \iff \bigcap_{i=1}^r \ker(\varphi_i) \subset \ker(\varphi)$.
\end{proposition}

\begin{remark}
    Soient $F, G$ deux sous-espaces vectoriels de $E$ (ou de $E^*$). Alors, on sait déjà que $F \subset G \iff G^\perp \subset F^\perp$. On a déjà montré le sens direct, et le sens inverse est vrai car si $G^\perp \subset F^\perp$, alors $(F^\perp)^\perp \subset (G^\perp)^\perp$, donc $F \subset G$, par la proposition \ref{prop:orthogonalite_double}.
\end{remark}

\begin{proof}
    On peut maintenant démontrer la proposition \ref{prop:orthogonalite_intersection} en utilisant la remarque précédente. En effet, on a :
    \begin{align*}
        \varphi \in \Vect(\varphi_1, \dots, \varphi_r) &\iff \Vect(\varphi) \subset \Vect(\varphi_1, \dots, \varphi_r) \\
        &\iff \left[\Vect(\varphi_1, \dots, \varphi_r)\right]^\perp \subset [\Vect(\varphi)]^\perp \\
        &\iff \bigcap_{i=1}^r \ker(\varphi_i) = \left[\Vect(\varphi_1, \dots, \varphi_r)\right]^\perp \subset [\Vect(\varphi)]^\perp = \ker(\varphi)
    \end{align*}
\end{proof}

\section{Equations d'un sous-espace vectoriel}

Soit $F$ un sous-espace vectoriel de $E$. On peut donner une équation de $F$ en utilisant l'orthogonalité. En effet, on peut écrire :
$$F = (F^\perp)^\perp$$
On pose alors $\varphi_1, \dots, \varphi_r$ une base de $F^\perp$. Alors :
$$F^\perp = \Vect(\varphi_1, \dots, \varphi_r)$$
$$\implies F = (F^\perp)^\perp = \left[\Vect(\varphi_1, \dots, \varphi_r)\right]^\perp = \bigcap_{i=1}^r \ker(\varphi_i)$$
Donc $F$ est l'intersection des noyaux de $\varphi_1, \dots, \varphi_r$. On peut donc donner une équation de $F$ en donnant une base de $F^\perp$.
$$x \in F \iff \begin{cases} 
\varphi_1(x) = 0 \\
\vdots \\
\varphi_r(x) = 0
\end{cases}$$
On dit que $(\varphi_1, \dots, \varphi_r)$ est un système d'équations de $F$.

De plus, $\codim F = \dim(F^\perp) = r$.

On réécrit ce passage sous forme de proposition.

\begin{proposition}
    Si $(\varphi_1,\dots,\varphi_r)$ est une base de $F^\perp$, alors :
    $$x \in F \iff \begin{cases}
    \varphi_1(x) = 0 \\
    \vdots \\
    \varphi_r(x) = 0
    \end{cases}$$
    On dit que $$\varphi_1(x) = 0, \dots, \varphi_r(x) = 0$$ est un système d'équations de $F$.
\end{proposition}

\begin{remark}
    Le système d'équations de $F$ n'est pas unique. En effet, si $(\varphi_1, \dots, \varphi_r)$ est une base de $F^\perp$, alors :
    $$\bigcap_{i=1}^r \ker\varphi_i = \bigcap_{j=1}^s \ker\psi_j
    \iff \Vect(\varphi_1, \dots, \varphi_r) = \Vect(\psi_1, \dots, \psi_s)$$
    \begin{align*}
        \text{En effet, } \bigcap_{i=1}^r \ker\varphi_i = \bigcap_{j=1}^s \ker\psi_j &\iff \left[\Vect(\varphi_1, \dots, \varphi_r)\right]^\perp = \left[\Vect(\psi_1, \dots, \psi_s)\right]^\perp \\
        &\iff \left[\Vect(\varphi_1, \dots, \varphi_r)\right]^{\perp\perp} = \left[\Vect(\psi_1, \dots, \psi_s)\right]^{\perp\perp} \\
        &\iff \Vect(\varphi_1, \dots, \varphi_r) = \Vect(\psi_1, \dots, \psi_s)
    \end{align*}
    Donc $\psi_1 = 0, \dots, \psi_s = 0$ est aussi un système d'équations de $F$.
\end{remark}
